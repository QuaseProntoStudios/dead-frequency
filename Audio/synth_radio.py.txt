"""Rádio 96.6 — cinco músicas originais geradas por código (sem direitos autorais de terceiros)."""
import sys, numpy as np
from scipy import signal
OUT = sys.argv[1]
SR = 32000
rng = np.random.default_rng(966)

def save(name, x):
    import wave
    x = np.asarray(x, dtype=np.float64)
    x = x / (np.max(np.abs(x)) or 1) * 0.9
    with wave.open(f"{OUT}/{name}.wav", "wb") as w:
        w.setnchannels(1); w.setsampwidth(2); w.setframerate(SR)
        w.writeframes((x * 32767).astype("<i2").tobytes())
    print(name, f"{len(x)/SR:.0f}s")

def lp(x, hz, o=2):
    b, a = signal.butter(o, min(hz / (SR / 2), 0.99), "low"); return signal.lfilter(b, a, x)
def hp(x, hz, o=2):
    b, a = signal.butter(o, hz / (SR / 2), "high"); return signal.lfilter(b, a, x)
def bp(x, lo, hi, o=2):
    b, a = signal.butter(o, [lo / (SR / 2), hi / (SR / 2)], "band"); return signal.lfilter(b, a, x)

def midi(n): return 440.0 * 2 ** ((n - 69) / 12)
def env(n, a=0.005, d=0.1, s=0.6, r=0.1):
    t = np.arange(n) / SR; total = n / SR
    e = np.where(t < a, t / max(a, 1e-4), s + (1 - s) * np.exp(-(t - a) / max(d, 1e-4)))
    rel = np.clip((total - t) / max(r, 1e-4), 0, 1)
    return e * rel

class Track:
    def __init__(self, seconds): self.buf = np.zeros(int(seconds * SR) + SR)
    def add(self, x, start, gain=1.0):
        s0 = int(start * SR); e = min(len(self.buf), s0 + len(x))
        if s0 < len(self.buf): self.buf[s0:e] += x[: e - s0] * gain

def osc(kind, f, dur, vib=0.0, vibf=5.5, detune=0.0):
    t = np.arange(int(dur * SR)) / SR
    ph = 2 * np.pi * np.cumsum(f * (1 + vib * np.sin(2 * np.pi * vibf * t)) * (1 + detune)) / SR
    if kind == "sine": return np.sin(ph)
    if kind == "tri": return 2 / np.pi * np.arcsin(np.sin(ph))
    if kind == "square": return np.sign(np.sin(ph)) * 0.6
    if kind == "saw":
        out = np.zeros_like(t)
        for k in range(1, 16): out += np.sin(k * ph) / k
        return out * 0.6
    if kind == "pluck":  # Karplus-Strong simplificado
        n = len(t); period = max(2, int(SR / f)); buf = rng.uniform(-1, 1, period); out = np.zeros(n)
        for i in range(n):
            out[i] = buf[i % period]; buf[i % period] = 0.5 * (buf[i % period] + buf[(i + 1) % period]) * 0.996
        return out
    if kind == "epiano":
        return (np.sin(ph) + 0.3 * np.sin(2 * ph) * np.exp(-t * 6) + 0.15 * np.sin(3 * ph) * np.exp(-t * 9))
    raise ValueError(kind)

def note(kind, n, dur, a=0.005, d=0.2, s=0.6, r=0.08, **kw):
    x = osc(kind, midi(n), dur, **kw); return x * env(len(x), a, d, s, r)

def kick(): t = np.arange(int(0.35 * SR)) / SR; f = 45 + 110 * np.exp(-t * 28); return np.sin(2 * np.pi * np.cumsum(f) / SR) * np.exp(-t * 8)
def snare(): n = int(0.22 * SR); t = np.arange(n) / SR; return (bp(rng.standard_normal(n), 1500, 7000) * 0.9 + np.sin(2 * np.pi * 185 * t) * 0.5) * np.exp(-t * 17)
def hat(o=False): n = int((0.2 if o else 0.05) * SR); t = np.arange(n) / SR; return hp(rng.standard_normal(n), 7000) * np.exp(-t * (12 if o else 70))
def rim(): n = int(0.05 * SR); t = np.arange(n) / SR; return (np.sin(2 * np.pi * 1700 * t) + 0.5 * bp(rng.standard_normal(n), 2000, 5000)) * np.exp(-t * 90)
def shaker(): n = int(0.08 * SR); t = np.arange(n) / SR; return bp(rng.standard_normal(n), 5000, 11000) * np.sin(np.pi * t / t[-1]) ** 2
def triangle(): n = int(0.4 * SR); t = np.arange(n) / SR; return sum(np.sin(2 * np.pi * f * t) for f in (3150, 4290, 5800)) * np.exp(-t * 6)
def chord_notes(root, kind):
    iv = {"maj7": [0, 4, 7, 11], "min7": [0, 3, 7, 10], "dom7": [0, 4, 7, 10], "maj": [0, 4, 7], "min": [0, 3, 7], "m9": [0, 3, 7, 10, 14], "6": [0, 4, 7, 9]}[kind]
    return [root + i for i in iv]

def master(x, lo=40, hi=12000, crackle=0.0):
    x = hp(lp(x, hi), lo)
    if crackle:
        c = (rng.random(len(x)) > 0.9993) * rng.uniform(-1, 1, len(x)); x = x + lp(c, 4000) * crackle + bp(rng.standard_normal(len(x)), 800, 5000) * crackle * 0.05
    return np.tanh(x / (np.max(np.abs(x)) or 1) * 1.4)

# 1) Frequência Morta — synthwave
def song1():
    bpm = 100; b = 60 / bpm; bars = 40; T = Track(bars * 4 * b + 2)
    prog = [(57, "min"), (53, "maj"), (48, "maj"), (55, "maj")]
    lead = [69, 72, 74, 76, 74, 72, 69, 67, 69, 72, 76, 79, 77, 76, 74, 72]
    for bar in range(bars):
        root, k = prog[bar % 4]; t0 = bar * 4 * b
        for nn in chord_notes(root, k): T.add(note("saw", nn, 4 * b, 0.4, 1.0, 0.7, 0.6, detune=0.003) + note("saw", nn, 4 * b, 0.4, 1.0, 0.7, 0.6, detune=-0.004), t0, 0.07)
        for i in range(16):
            T.add(note("saw", root - 12 + (12 if i % 2 else 0), b / 4, 0.002, 0.08, 0.3, 0.03), t0 + i * b / 4, 0.18)
        if bar >= 4:
            for i in (0, 2): T.add(kick(), t0 + i * b * 1.0 + (b if i else 0), 0.9)
            T.add(kick(), t0 + 2 * b, 0.9)
            for i in (1, 3): T.add(snare(), t0 + i * b, 0.45)
            for i in range(8): T.add(hat(i == 7), t0 + i * b / 2, 0.12)
        if 8 <= bar < 36:
            for i in range(4):
                n = lead[(bar * 4 + i) % 16]
                T.add(note("square", n, b * 0.9, 0.01, 0.3, 0.5, 0.1, vib=0.004), t0 + i * b, 0.16)
    return master(T.buf)

                T.add(note("pluck", nn - 12, 0.9 * b), t0 + p * b + j * 0.012, 0.18)
        for i in (0, 1.5, 3, 4.5, 6, 7) if False else (0.0, 0.75, 1.5, 2.5, 3.0, 3.5):

# 3) Asfalto Molhado — lo-fi hip hop
def song3():
    bpm = 84; b = 60 / bpm; bars = 32; T = Track(bars * 4 * b + 2)
    prog = [(60, "m9"), (65, "dom7"), (58, "maj7"), (63, "maj7")]
    for bar in range(bars):
        root, k = prog[bar % 4]; t0 = bar * 4 * b
        for j, nn in enumerate(chord_notes(root, k)):
            T.add(note("epiano", nn - 12, 3.6 * b, 0.01, 1.5, 0.3, 0.5), t0 + j * 0.02, 0.11)
        T.add(note("sine", root - 24, 1.8 * b, 0.01, 0.6, 0.6, 0.2), t0, 0.5); T.add(note("sine", root - 24 + 7, 1.6 * b, 0.01, 0.6, 0.6, 0.2), t0 + 2.5 * b, 0.4)
        sw = 0.06 * b
        for i in (0, 1.75, 2.5): T.add(kick(), t0 + i * b, 0.8)
        for i in (1, 3): T.add(snare(), t0 + i * b + 0.02, 0.4)
        for i in range(8): T.add(hat(), t0 + i * b / 2 + (sw if i % 2 else 0), 0.08)
        if bar % 4 == 2:
            for i, n in enumerate([75, 72, 70, 67]): T.add(note("tri", n, 0.8 * b, 0.02, 0.3, 0.4, 0.2), t0 + i * b * 0.75, 0.1)
    return master(T.buf, hi=6500, crackle=0.25)

# 4) Forró do Entregador — sanfona, zabumba e triângulo
def song4():
    bpm = 124; b = 60 / bpm; bars = 44; T = Track(bars * 4 * b + 2)
    prog = [(55, "maj"), (60, "maj"), (62, "dom7"), (55, "maj"), (55, "maj"), (60, "maj"), (62, "dom7"), (55, "maj")]
    scale = [55, 57, 59, 60, 62, 64, 65, 67, 69, 71, 72, 74]
    melody = [7, 8, 9, 7, 5, 4, 5, 7, 4, 2, 4, 5, 4, 2, 0, 2]
    for bar in range(bars):
        root, k = prog[bar % 8]; t0 = bar * 4 * b
        for p in (0.5, 1.5, 2.5, 3.5):  # sanfona no contratempo
            for nn in chord_notes(root, k): T.add(note("saw", nn, 0.4 * b, 0.01, 0.15, 0.6, 0.05, vib=0.006, vibf=6), t0 + p * b, 0.05)
        T.add(note("sine", root - 24, b * 0.9, 0.005, 0.3, 0.5, 0.05), t0, 0.5); T.add(note("sine", root - 17, b * 0.9, 0.005, 0.3, 0.5, 0.05), t0 + 2 * b, 0.45)
        for i in (0, 1.5, 2): T.add(kick(), t0 + i * b, 0.75)
        for i in (1, 3): T.add(snare(), t0 + i * b, 0.25)
        for i in range(8): T.add(triangle(), t0 + i * b / 2, 0.05 if i % 2 == 0 else 0.1)
        if 4 <= bar < 40:
            for i in range(4):
                n = scale[melody[(bar * 4 + i) % 16]] + 12
                T.add(note("saw", n, b * 0.85, 0.02, 0.2, 0.7, 0.08, vib=0.008, vibf=6.5), t0 + i * b, 0.13)
    return master(T.buf, hi=9000)

# 5) Neon na Rodovia — chiptune rápido
def song5():
    bpm = 148; b = 60 / bpm; bars = 56; T = Track(bars * 4 * b + 2)
    prog = [(64, "min"), (60, "maj"), (67, "maj"), (62, "maj")]
    lead = [76, 79, 83, 79, 81, 79, 76, 74, 76, 79, 84, 83, 81, 79, 81, 83]
    for bar in range(bars):
        root, k = prog[bar % 4]; t0 = bar * 4 * b; ns = chord_notes(root, k)
        for i in range(16): T.add(note("square", ns[i % 3] + 12, b / 4, 0.001, 0.05, 0.4, 0.01), t0 + i * b / 4, 0.07)
        for i in range(8): T.add(note("tri", root - 12 + (0 if i % 2 == 0 else 12), b / 2, 0.002, 0.1, 0.6, 0.02), t0 + i * b / 2, 0.35)
        if bar >= 2:
            for i in range(4): T.add(kick(), t0 + i * b, 0.7)
            for i in (1, 3): T.add(snare(), t0 + i * b, 0.35)
            for i in range(8): T.add(hat(), t0 + i * b / 2 + b / 4, 0.08)
        if 8 <= bar < 52:
            for i in range(8):
                n = lead[(bar * 8 + i) % 16]
                if (bar + i) % 5 != 4: T.add(note("square", n, b / 2 * 0.9, 0.003, 0.1, 0.6, 0.03, vib=0.003), t0 + i * b / 2, 0.13)
    return master(T.buf)

# Vinheta da rádio (entre as músicas)
def jingle():
    T = Track(3.0)
    for i, n in enumerate([72, 76, 79, 84]): T.add(note("sine", n, 0.6, 0.01, 0.3, 0.3, 0.2) + note("tri", n + 12, 0.6, 0.01, 0.2, 0.2, 0.2) * 0.3, i * 0.14, 0.4)
    t = np.arange(int(1.2 * SR)) / SR
    T.add(np.sin(2 * np.pi * 966 * t) * np.exp(-t * 3) * 0.15, 0.8)
    T.add(bp(rng.standard_normal(int(0.5 * SR)), 1500, 4000) * np.linspace(0.3, 0, int(0.5 * SR)), 1.9)
    return master(T.buf[: int(2.6 * SR)])

songs = [("radio_01_frequencia_morta", song1), ("radio_02_bossa_da_madrugada", song2), ("radio_03_asfalto_molhado", song3),
         ("radio_04_forro_do_entregador", song4), ("radio_05_neon_na_rodovia", song5)]
import sys as _s
only = _s.argv[2:] or None
for name, fn in songs:
    if only and name not in only: continue
    save(name, fn())
if not only: save("radio_jingle", jingle())
