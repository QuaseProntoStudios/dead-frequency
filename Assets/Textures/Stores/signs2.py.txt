def fascia(name, bg, fg, text, stripe):
    W2, H2 = 2048, 128
    img = Image.new("RGB", (W2, H2), bg); d = ImageDraw.Draw(img)
    d.rectangle((0, H2 - 22, W2, H2), fill=stripe)
    f, b = fit(d, text, D + "DejaVuSans-BoldOblique.ttf", 900, 86)
    for cx in (W2 // 4, 3 * W2 // 4):
        d.text((cx - (b[2] - b[0]) // 2 - b[0], 8 - b[1] + (86 - (b[3] - b[1])) // 2), text, font=f, fill=fg)
    img.filter(ImageFilter.SMOOTH).save(f"{OUT}/fascia_{name}.png")

def totem(name, bg, fg, brand):

def ad(name, bg, title, title_color, line, line_color, painter=None, title_font=D + "DejaVuSans-Bold.ttf"):
    W2, H2 = 1200, 400
    img = Image.new("RGB", (W2, H2), bg); d = ImageDraw.Draw(img)
    if painter: painter(d, W2, H2)
    f, b = fit(d, title, title_font, 860, 150)
    x = (W2 - (b[2] - b[0])) // 2 - b[0]; y = 60 - b[1] + (150 - (b[3] - b[1])) // 2
    d.text((x + 6, y + 6), title, font=f, fill=(0, 0, 0))
    d.text((x, y), title, font=f, fill=title_color)
    f2, b2 = fit(d, line, D + "DejaVuSansCondensed-Bold.ttf", 840, 64)
    d.text(((W2 - (b2[2] - b2[0])) // 2 - b2[0], 285 - b2[1]), line, font=f2, fill=line_color)
    d.rectangle((0, 0, W2 - 1, H2 - 1), outline=(240, 240, 240), width=10)
    img.filter(ImageFilter.SMOOTH).save(f"{ADS}/ad_{name}.png")

def p_waves(d, W2, H2):
    for i in range(6):
        pts = [(x, 330 + 14 * math.sin(x / 40 + i) - i * 4) for x in range(0, W2, 8)]
        d.line(pts, fill=(40, 160 - i * 15, 70), width=3)
def p_checker(d, W2, H2):
    for x in range(0, W2, 80):
        for y in range(0, H2, 80):
            if (x // 80 + y // 80) % 2 == 0: d.rectangle((x, y, x + 80, y + 80), fill=(200, 30, 30))
    d.rectangle((60, 40, W2 - 60, 360), fill=(250, 245, 235))
